cmake_minimum_required(VERSION 3.24)
project(QvtuCampusLogin VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(WEBVIEW2_SDK "" CACHE PATH "Path to WebView2 Native SDK build/native directory")
if(NOT WEBVIEW2_SDK AND DEFINED ENV{WEBVIEW2_SDK})
  set(WEBVIEW2_SDK "$ENV{WEBVIEW2_SDK}")
endif()
if(NOT WEBVIEW2_SDK AND DEFINED ENV{NUGET_PACKAGES})
  file(GLOB _wv2_candidates "$ENV{NUGET_PACKAGES}/microsoft.web.webview2/*/build/native")
  if(_wv2_candidates)
    list(SORT _wv2_candidates COMPARE NATURAL ORDER DESCENDING)
    list(GET _wv2_candidates 0 WEBVIEW2_SDK)
  endif()
endif()
if(NOT WEBVIEW2_SDK)
  set(_default_nuget "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2")
  if(EXISTS "${_default_nuget}")
    file(GLOB _wv2_candidates "${_default_nuget}/*/build/native")
    if(_wv2_candidates)
      list(SORT _wv2_candidates COMPARE NATURAL ORDER DESCENDING)
      list(GET _wv2_candidates 0 WEBVIEW2_SDK)
    endif()
  endif()
endif()
if(NOT WEBVIEW2_SDK OR NOT EXISTS "${WEBVIEW2_SDK}/include/WebView2.h" OR NOT EXISTS "${WEBVIEW2_SDK}/x64/WebView2LoaderStatic.lib")
  message(FATAL_ERROR "WebView2 Native SDK not found. Install Microsoft.Web.WebView2 NuGet package, then set -DWEBVIEW2_SDK=<package>/build/native or WEBVIEW2_SDK environment variable.")
endif()

add_executable(QvtuSettings WIN32 src/settings.cpp src/config.cpp src/wifi.cpp src/config.h src/wifi.h)
add_executable(QvtuLogin WIN32 src/login.cpp src/config.cpp src/wifi.cpp src/config.h src/wifi.h)
set_target_properties(QvtuSettings PROPERTIES OUTPUT_NAME "QVTLogin_Config")
set_target_properties(QvtuLogin PROPERTIES OUTPUT_NAME "QVTLogin")
target_include_directories(QvtuSettings PRIVATE "${WEBVIEW2_SDK}/include" src)
target_include_directories(QvtuLogin PRIVATE "${WEBVIEW2_SDK}/include" src)
target_link_libraries(QvtuSettings PRIVATE "${WEBVIEW2_SDK}/x64/WebView2LoaderStatic.lib" advapi32 crypt32 shlwapi comctl32 wlanapi ole32)
target_link_libraries(QvtuLogin PRIVATE "${WEBVIEW2_SDK}/x64/WebView2LoaderStatic.lib" advapi32 crypt32 shlwapi comctl32 ole32 wlanapi)
target_compile_definitions(QvtuSettings PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_definitions(QvtuLogin PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
  target_compile_options(QvtuSettings PRIVATE /W4 /utf-8)
  target_compile_options(QvtuLogin PRIVATE /W4 /utf-8)
endif()

install(TARGETS QvtuSettings RUNTIME DESTINATION .)
install(TARGETS QvtuLogin RUNTIME DESTINATION payload)
